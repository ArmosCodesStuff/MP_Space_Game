// drops this in as scripts/_Shots.cs, registers it as an autoload, and runs the real
// renderer on a virtual display. Frames land in /tmp/shots/. Stored as .txt so
// nothing compiles it in place. Note: on a real display Godot reads the real
// pointer, so parsed mouse-motion events do not move the aim here.

    // ── UI lint: run on every frame ──────────────────────────────────────────
    // off-screen   : a visible control not fully inside its viewport (scroll contents exempt)
    // clipped      : a label or button whose text is wider than the box it is drawn in
    // overlap      : two top-level HUD blocks (direct children of a CanvasLayer) overlapping
    static bool UnderScroll(Node n) { for (var p = n.GetParent(); p != null; p = p.GetParent()) if (p is ScrollContainer) return true; return false; }
    static float TextW(Control c, string text)
    {
        var f = c.GetThemeFont("font"); int fs = c.GetThemeFontSize("font_size");
        float w = 0; foreach (var line in text.Split('\n')) w = Mathf.Max(w, f.GetStringSize(line, HorizontalAlignment.Left, -1, fs).X);
        return w;
    }
    void Lint(string frame)
    {
        var tops = new System.Collections.Generic.List<Control>();
        foreach (var c in N<Control>())
        {
            if (!c.IsVisibleInTree()) continue;
            var r = c.GetGlobalRect(); var vp = c.GetViewportRect();
            if (r.Size.X < 1 || r.Size.Y < 1) continue;
            string id = $"{c.GetType().Name} '{c.Name}'" + (c is Label l0 && l0.Text.Length > 0 ? $" \"{l0.Text.Split('\n')[0][..Mathf.Min(28, l0.Text.Split('\n')[0].Length)]}\"" : c is Button b0 && b0.Text.Length > 0 ? $" [{b0.Text[..Mathf.Min(24, b0.Text.Length)]}]" : "");
            if (!UnderScroll(c) && !(r.Position.X >= vp.Position.X - 1 && r.Position.Y >= vp.Position.Y - 1 && r.End.X <= vp.End.X + 1 && r.End.Y <= vp.End.Y + 1))
                GD.Print($"LINT {frame}: off-screen {id} at {r.Position.Round()} size {r.Size.Round()}");
            if (c is Label lb && lb.AutowrapMode == TextServer.AutowrapMode.Off && !lb.ClipText && lb.Text.Length > 0 && TextW(lb, lb.Text) > r.Size.X + 2)
                GD.Print($"LINT {frame}: clipped {id} needs {TextW(lb, lb.Text):0} px, has {r.Size.X:0}");
            if (c is Button bt && bt.Text.Length > 0 && !bt.ClipText && TextW(bt, bt.Text) + 8 > r.Size.X + 2)
                GD.Print($"LINT {frame}: clipped {id} needs {TextW(bt, bt.Text) + 8:0} px, has {r.Size.X:0}");
            // (the hint card takes no input, but it is a block on the HUD all the same)
            if (c.GetParent() is CanvasLayer && !(c is ColorRect) && c.MouseFilter != Control.MouseFilterEnum.Ignore || c is HullHud || c is AbilityBar || c is BasePanel || c.GetParent() is Hints)
                if (c.GetParent() is CanvasLayer) tops.Add(c);
        }
        for (int i = 0; i < tops.Count; i++)
            for (int j = i + 1; j < tops.Count; j++)
            {
                var a = tops[i].GetGlobalRect(); var b = tops[j].GetGlobalRect();
                var x = a.Intersection(b);
                if (x.Size.X > 4 && x.Size.Y > 4 && tops[i].GetViewport() == tops[j].GetViewport())
                    GD.Print($"LINT {frame}: overlap {tops[i].GetType().Name} '{tops[i].Name}' / {tops[j].GetType().Name} '{tops[j].Name}' ({x.Size.X:0}x{x.Size.Y:0})");
            }
    }
    Hub H => GetTree().CurrentScene as Hub;
    System.Collections.Generic.IEnumerable<T> N<T>(Node n = null) where T : Node { n ??= GetTree().Root;
        foreach (var c in n.GetChildren()) { if (c is T t) yield return t; foreach (var x in N<T>(c)) yield return x; } }
    void Key(Key k) { Input.ParseInputEvent(new InputEventKey { Keycode = k, PhysicalKeycode = k, Pressed = true }); Input.ParseInputEvent(new InputEventKey { Keycode = k, PhysicalKeycode = k, Pressed = false }); }
    object P(object o, string m, params object[] a) => o.GetType().GetMethod(m, BindingFlags.NonPublic | BindingFlags.Instance).Invoke(o, a);
