    // The loudest sample in a wav, 0..1. Reads the asset itself: a check on a level that reads a
    // number in the code would pass whatever the file contains.
    static double Peak(string path)
    {
        // THE FILE, not GD.Load. An imported AudioStreamWav is what Godot made of the asset --
        // the importer can trim, loop or resample it -- so measuring that answers a question
        // about the importer. The claim here is about the asset as shipped, so read the bytes.
        var d = FileAccess.GetFileAsBytes(path);
        if (d == null || d.Length < 45) return 0;
        int max = 0;
        for (int i = 44; i + 1 < d.Length; i += 2)     // 44: past a canonical PCM WAV header
        {
            // Sign-extended and negated by hand: Math.Abs picks its short overload here, and
            // Math.Abs(short.MinValue) throws rather than returning 32768.
            int v = d[i] | (d[i + 1] << 8);
            if (v > 32767) v -= 65536;
            if (v < 0) v = -v;
            if (v > max) max = v;
        }
        return max / 32768.0;
    }

    // Everything Character holds, as one string, ENUMERATED BY REFLECTION. A list written out
    // here by hand would be a list that stops being true the first time someone adds a field.
    static string Describe(object v) => v switch
    {
        null => "null",
        string str => str,
        System.Collections.IDictionary d => "{" + string.Join(",", d.Keys.Cast<object>()
            .OrderBy(k => k.ToString(), StringComparer.Ordinal).Select(k => k + ":" + Describe(d[k]))) + "}",
        // An array's ORDER is part of its value (Bought is indexed by upgrade); a set's is not.
        Array arr => "[" + string.Join(",", arr.Cast<object>().Select(Describe)) + "]",
        System.Collections.IEnumerable e => "[" + string.Join(",", e.Cast<object>()
            .Select(Describe).OrderBy(x => x, StringComparer.Ordinal)) + "]",
        _ => Convert.ToString(v, System.Globalization.CultureInfo.InvariantCulture),
    };

    // NonPublic too: Spares is private static readonly, so a public-only sweep never saw it at
    // all -- neither the dump nor the inventory. Persisted state does not have to be public.
    const BindingFlags CharFields = BindingFlags.Public | BindingFlags.NonPublic | BindingFlags.Static;

    static string DumpCharacter() => string.Join(";", typeof(Character)
        .GetFields(CharFields)
        .OrderBy(f => f.Name, StringComparer.Ordinal)
        .Select(f => f.Name + "=" + Describe(f.GetValue(null))));

    // ── the save carries everything the character IS, and only this build can read it ────────
    async Task SaveCoverage()
    {
        // Every class is touched first: Load() fills the loadout and spares for ALL of them, so a
        // character that had only ever used one would differ after a round trip for that reason
        // alone and say nothing about saving.
        Character.NewBlank();
        foreach (ShipClass sc in Enum.GetValues(typeof(ShipClass))) { Character.LoadoutFor(sc); Character.SparesFor(sc); }
        Character.Name = "Roundtrip"; Character.Class = ShipClass.Carrier;
        Character.Main = new Color(0.11f, 0.22f, 0.33f); Character.Accent = new Color(0.44f, 0.55f, 0.66f);
        Character.Exp = 777; Character.Level = 5; Character.Points = 3;
        for (int i = 0; i < Character.Bought.Length; i++) Character.Bought[i] = i % (Progression.MaxPerUpgrade + 1);
        Character.BossCleared["ironhold"] = new System.Collections.Generic.HashSet<int> { 1, 3 };
        Character.BaseOre = 123.5; Character.BaseSalvage = 67.25; Character.BaseCredits = 4321;
        Character.BaseLevels[Economy.All[0].Id] = 3;
        Character.BaseInvested[Economy.All[0].Tab] = 900;
        Character.SparesFor(ShipClass.Carrier).Add(Equipment.All.First(e => e.Slot == GearSlot.Chip).Id);
        Character.Bonuses["hull"] = 0.25;
        string id = Character.Id;
        Character.Save(); await Wait(0.05);
        string before = DumpCharacter();
        Character.NewBlank();                                  // wipe it out of memory entirely
        bool loaded = Character.Load(id);
        string after = DumpCharacter();
        // A FIELD ADDED TO Character AND NOT TO Save() FAILS HERE, on the day it is added. That is
        // what "the tests adapt to what changed" has to mean to be worth anything: not a promise
        // to remember, a check that notices.
        Check(loaded && before == after,
              loaded ? (before == after ? "everything a character holds survives a save and a load"
                                        : $"a character field is not being saved -- before {before}  |  after {after}")
                     : "everything a character holds survives a save and a load (the load failed)");

        // THE ROUND TRIP ONLY COVERS WHAT IT SETS, and reflection over the dump does not change
        // that: a field left at its default is identical before and after whether it is saved or
        // not. This is the part that actually notices -- every public static field of Character,
        // against the list this test exercises. Add a field and this goes red until it is either
        // set above or named here as deliberately not persisted. Found the hard way: the base
        // economy fields were added, not saved, and the round trip passed.
        var exercised = new[] { "Accent", "BaseCredits", "BaseInvested", "BaseLevels", "BaseOre",
                                "BaseSalvage", "BossCleared", "Bonuses", "Bought", "Class", "Exp",
                                "Id", "Level", "Loadout", "Main", "Name", "Points", "Spares", "Version" };
        // Not state a character carries: a path, and the compiler's backing field for a property
        // that reports how the LAST save went. Named explicitly rather than filtered by a pattern,
        // so a real field that happens to look like one cannot slip through.
        var notPersisted = new[] { "Dir", "<LastSaveRenamed>k__BackingField" };
        var present = typeof(Character).GetFields(CharFields)
                                       .Select(f => f.Name).OrderBy(n => n, StringComparer.Ordinal).ToArray();
        var accounted = exercised.Concat(notPersisted).OrderBy(n => n, StringComparer.Ordinal).ToArray();
        Check(present.SequenceEqual(accounted),
              $"every field of Character is accounted for by this test (new: {string.Join(",", present.Except(accounted))}; gone: {string.Join(",", accounted.Except(present))})");

        // ── the build stamp ──
        Check(Character.List().First(x => x.Id == id).Version == 1 && Game.Version == 1,
              $"a character records the build that wrote it (version {Character.List().First(x => x.Id == id).Version}, game {Game.Version})");

        // A character from ANOTHER build: written by hand on disk, because the only other way to
        // make one is to be another build.
        var cfg = new ConfigFile();
        Check(cfg.Load($"{Character.Dir}/{id}.cfg") == Error.Ok, "the saved character reads back as a config file");
        cfg.SetValue("id", "version", Game.Version + 1);
        cfg.SetValue("id", "name", "FromAnotherBuild");
        string alienId = "alien";
        cfg.Save($"{Character.Dir}/{alienId}.cfg");
        await Wait(0.05);
        var alien = Character.List().FirstOrDefault(x => x.Id == alienId);
        Check(alien != null && !alien.Playable && alien.Name == "FromAnotherBuild",
              "a character from another build is listed, and marked unplayable");
        string keptId = Character.Id;
        Check(!Character.Load(alienId) && Character.Id == keptId,
              $"loading it is refused, and it does not become the character the next save overwrites ({Character.Id})");

        // ...and the select screen greys it out and says why, rather than failing on PLAY.
        GetTree().ChangeSceneToFile("res://CharacterSelect.tscn"); await Wait(0.3);
        var sel = GetTree().CurrentScene as CharacterSelect;
        var row = Nodes<PanelContainer>(sel).FirstOrDefault(p => p.Name == "Row_" + alienId);
        var play = row == null ? null : Nodes<Button>(row).FirstOrDefault(b => b.Name == "Play");
        var note = row == null ? null : Nodes<Label>(row).FirstOrDefault(l => l.Name == "Incompatible");
        Check(row != null && play != null && play.Disabled && note != null && note.Text == Game.IncompatibleNote,
              $"the row is greyed out, PLAY is dead, and it says why ({note?.Text})");
        // The playable one beside it is still playable: this greys the character that cannot be
        // read, not the screen.
        var mineRow = Nodes<PanelContainer>(sel).FirstOrDefault(p => p.Name == "Row_" + id);
        var minePlay = mineRow == null ? null : Nodes<Button>(mineRow).FirstOrDefault(b => b.Name == "Play");
        Check(minePlay != null && !minePlay.Disabled, "the character from THIS build is still playable");

        Character.Delete(alienId); Character.Delete(id);
        await Wait(0.05);
    }

    // ── the title screen ──────────────────────────────────────────────────────
    // The menu used to be sprites and structs, self-contained so no gameplay change could break
    // it -- and the cost was that it could show something the game does not do. It flies the REAL
    // battleship now, so these checks are also a check on the battleship.
    async Task MenuDiorama()
    {
        var menu = GetTree().CurrentScene as MainMenu;
        Check(menu?.DemoShip != null && menu.DemoShip.Demo && menu.DemoShip.Class == ShipClass.Battleship,
              "the title screen flies a REAL battleship, not a picture of one");
        if (menu?.DemoShip == null) return;
        var ship = menu.DemoShip;
        // The spec's numbers as literals, never the menu's own constants: a check that reads the
        // constant cannot catch the constant being wrong.
        Check(Math.Abs(ship.Stats["missile_mag"] - 2) < 1e-9 && Math.Abs(ship.Stats["missile_reload"] - 5) < 1e-9
              && Mathf.IsEqualApprox(ship.WarpHop, 500f),
              $"two missiles on a 5 s reload and a 500 u hop ({ship.Stats["missile_mag"]:0} / {ship.Stats["missile_reload"]:0} s / {ship.WarpHop:0} u)");
        // The ship has to be the SIZE it is in game. The menu had its own zoom for a while and the
        // battleship was half as big again on the title screen as it is the moment you press PLAY.
        // Compared with 0.9 as a literal rather than with Hub.DefaultZoom, so this also fails if
        // the game's zoom is retuned and the menu is left behind.
        Check(Mathf.IsEqualApprox(menu.CameraZoom, 0.9f) && Mathf.IsEqualApprox(menu.CameraZoom, Hub.DefaultZoom),
              $"the title screen shows the ship at the game's own zoom ({menu.CameraZoom:0.00})");
        // ...and the ENEMIES are the size they are in the game too, for the same reason. Compared
        // against the raiders' own lengths: the claim is equality WITH THE GAME, not a magnitude,
        // so the game's number is exactly the right thing to compare with.
        Check(menu.Foes.Where(f => f.Kind == MenuFoeKind.Heavy).All(f => Mathf.IsEqualApprox(f.HitRadius, Raider.HeavyLength * 0.3f))
              && menu.Foes.Where(f => f.Kind != MenuFoeKind.Heavy).All(f => Mathf.IsEqualApprox(f.HitRadius, Raider.LightLength * 0.4f)),
              "the title screen's enemies are the size the game's raiders are");
        Check(menu.Foes.Count(f => f.Kind == MenuFoeKind.Heavy) == 2
              && menu.Foes.Count(f => f.Kind == MenuFoeKind.Web) == 1
              && menu.Foes.Count(f => f.Kind == MenuFoeKind.Light) == 3
              && menu.Foes.All(f => Combat.Hostiles.Contains(f)),
              "heavies, a webifier and light fighters, all of them real targets the ship can shoot");
        // HOLDING STATION. Pushed well off its mark first: a ship that was never going to move
        // cannot demonstrate that it comes back, which is the same vacuity trap as the boss lock.
        ship.Position = menu.Station + new Vector2(200, 150);
        double drift0 = ship.Position.DistanceTo(menu.Station);
        await Wait(8.0);
        double drift1 = ship.Position.DistanceTo(menu.Station);
        Check(drift0 > 200 && drift1 < 120,
              $"pushed {drift0:0} u off station it flies itself home ({drift1:0} u)");

        // THE DODGE. An area shot lands in 8 s; it starts the jump with 4 to go and the warp takes
        // 3, so it is clear ONE SECOND before impact. Each number is checked, not just the last.
        menu.ForceAreaShot();
        // Wait for the shot to actually be CALLED. Reading the countdown one frame after asking
        // for it read -1 (no shot yet) and every measurement below was taken from that -1.
        for (int i = 0; i < 60 && menu.AreaShotIn < 0; i++) await ToSignal(GetTree(), SceneTree.SignalName.ProcessFrame);
        double called = menu.AreaShotIn;
        double startedAt = -1, landedAt = -1;
        var before = ship.Position;
        for (int i = 0; i < 60 * 12 && menu.AreaShotIn > 0; i++)
        {
            if (startedAt < 0 && ship.Warping) startedAt = menu.AreaShotIn;
            if (startedAt >= 0 && landedAt < 0 && !ship.Warping) landedAt = menu.AreaShotIn;
            await ToSignal(GetTree(), SceneTree.SignalName.ProcessFrame);
        }
        Check(Math.Abs(called - 8) < 0.2, $"the area shot is called 8 s before it lands ({called:0.0} s)");
        Check(startedAt > 3.0 && startedAt <= 4.05, $"it starts the jump with four seconds to go ({startedAt:0.0} s)");
        Check(landedAt > 0.6 && landedAt < 1.6, $"and is clear a whole second before impact ({landedAt:0.0} s)");
        Check(before.DistanceTo(ship.Position) > 300, $"the jump actually moved it ({before.DistanceTo(ship.Position):0} u)");
    }

    // ── character select ─────────────────────────────────────────────────────
    async Task Select()
    {
        GetTree().ChangeSceneToFile("res://CharacterSelect.tscn");
        await Wait(0.3);
        var sel = GetTree().CurrentScene as CharacterSelect;
        Check(sel != null, "select screen loads");
        // EVERY RUN STARTS WITH NOBODY. run.ps1 deletes the whole user:// folder first, so this
        // has always been true -- but nothing said so, and a run that quietly reused yesterday's
        // character would look like a passing run with an unexplained failure in it somewhere.
        // Asserted here, before anything creates one.
        Check(Character.List().Count == 0, $"this run starts with no saved characters ({Character.List().Count})");
        var slots = Character.List();
        // start with one saved character, made the way a player would
        Character.NewBlank(); Character.Name = "OldTimer"; Character.Class = ShipClass.Carrier; Character.Save();
        await Wait(0.05);
        GetTree().ChangeSceneToFile("res://CharacterSelect.tscn"); await Wait(0.3);
        sel = GetTree().CurrentScene as CharacterSelect;
        slots = Character.List();
        Check(slots.Count == 1 && slots[0].Name == "OldTimer", "a saved character is listed");
        Check(Nodes<PanelContainer>(sel).Count(p => p.Name.ToString().StartsWith("Row_")) == 1, "one row shown");

        // Character.Save writes a temp file and renames it over the real one, so a second
        // instance sharing this user:// can never read a half-written character. Saving
        // straight onto the live path caused "ConfigFile parse error ... Unterminated
        // string" on roughly 1 smoke run in 3. Two consequences are observable from here:
        // a stray .tmp is not a character, and the next save consumes it.
        var tmpPath = $"{Character.Dir}/{Character.Id}.cfg.tmp";
        using (var stray = FileAccess.Open(tmpPath, FileAccess.ModeFlags.Write))
            stray.StoreString("[id]\nname=\"half written");
        Check(FileAccess.FileExists(tmpPath) && Character.List().Count == 1,
              "a half-written .tmp is never listed as a character");
        Character.Save(); await Wait(0.05);
        Check(!FileAccess.FileExists(tmpPath),
              "Save() goes through a temp file and renames it away, leaving none behind");
        // ...and it really RENAMED rather than falling back to writing over the live file. Both
        // paths leave the same thing on disk, so the check above passes either way; only this one
        // can tell. Windows' rename() refuses an existing destination -- Godot works around that,
        // but if it ever stopped, every save would quietly take the unsafe path.
        Check(Character.LastSaveRenamed,
              "the save was committed by the rename, not by the fallback that writes over the live file");
        // user://settings.cfg is written the same way and for the same reason: two instances share
        // one user://, so a rebind saved by one used to leave the other reading a torn file and
        // silently falling back to defaults.
        var setTmp = Settings.Path + ".tmp";
        using (var strayCfg = FileAccess.Open(setTmp, FileAccess.ModeFlags.Write))
            strayCfg.StoreString("[audio]\nvolume_idx=");
        Settings.Save(); await Wait(0.05);
        Check(!FileAccess.FileExists(setTmp) && Settings.LastSaveRenamed,
              "Settings.Save() goes through a temp file too, and commits it by the rename");

        // new character: cancel writes nothing
        Nodes<Button>(sel).First(b => b.Text == "NEW CHARACTER").EmitSignal(BaseButton.SignalName.Pressed);
        await Wait(0.1);
        var cc = Nodes<CharacterCreator>().First();
        Check(cc.IsNew, "NEW CHARACTER opens a creator in new mode");
        Key(global::Godot.Key.Escape); await Wait(0.1);
        Check(Count<CharacterCreator>() == 0 && Character.List().Count == 1, "Esc cancels a new character without saving");

        // new character: create
        Nodes<Button>(sel).First(b => b.Text == "NEW CHARACTER").EmitSignal(BaseButton.SignalName.Pressed);
        await Wait(0.1);
        cc = Nodes<CharacterCreator>().First();
        var name = Nodes<LineEdit>(cc).First(); name.Text = "Ironside"; name.EmitSignal(LineEdit.SignalName.TextChanged, "Ironside");
        Nodes<Button>(cc).First(b => b.Text.StartsWith("BATTLESHIP")).EmitSignal(BaseButton.SignalName.Pressed);
        await Wait(0.05);
        Nodes<Button>(cc).First(b => b.Text == "CREATE").EmitSignal(BaseButton.SignalName.Pressed);
        await Wait(0.2);
        slots = Character.List();
        Check(slots.Count == 2 && slots.Any(s => s.Name == "Ironside" && s.Class == ShipClass.Battleship), "CREATE saves a second character");
        Check(Nodes<PanelContainer>(sel).Count(p => p.Name.ToString().StartsWith("Row_")) == 2, "list shows both");

        // delete: KEEP leaves it, DELETE removes the file
        var old = slots.First(s => s.Name == "OldTimer");
        var row = Nodes<PanelContainer>(sel).First(p => p.Name == "Row_" + old.Id);
        Nodes<Button>(row).First(b => b.Name == "Delete").EmitSignal(BaseButton.SignalName.Pressed);
        await Wait(0.1);
        var dlg = Nodes<ConfirmationDialog>(sel).First();
        Check(dlg.Visible && dlg.DialogText.Contains("OldTimer"), "DELETE asks first, naming the character");
        dlg.GetCancelButton().EmitSignal(BaseButton.SignalName.Pressed); await Wait(0.1);
        Check(Character.List().Count == 2, "KEEP deletes nothing");
        row = Nodes<PanelContainer>(sel).First(p => p.Name == "Row_" + old.Id);
        Nodes<Button>(row).First(b => b.Name == "Delete").EmitSignal(BaseButton.SignalName.Pressed);
        await Wait(0.1);
        dlg.GetOkButton().EmitSignal(BaseButton.SignalName.Pressed); await Wait(0.2);
        Check(Character.List().Count == 1 && !FileAccess.FileExists($"{Character.Dir}/{old.Id}.cfg"), "confirmed DELETE removes the file");
        Check(Nodes<PanelContainer>(sel).Count(p => p.Name.ToString().StartsWith("Row_")) == 1, "row gone from the list");

        // play
        var iron = Character.List().First();
        row = Nodes<PanelContainer>(sel).First(p => p.Name == "Row_" + iron.Id);
        Nodes<Button>(row).First(b => b.Name == "Play").EmitSignal(BaseButton.SignalName.Pressed);
        await Wait(0.4);
        Check(H != null && Character.Name == "Ironside", "PLAY enters the hub as that character");
    }

    // MY ship on every peer: "Ship_1" is only right offline and on the host (id 1);

        var yard = H.Yard;

    }

    // ── multiplayer ──────────────────────────────────────────────────────────
    // ── a dedicated two-player arena run (its own port, after the three-player run) ──
    // Mission 1: both READY, both at the portal, into the arena together; the boss dies;

    async Task ArenaMp(string role)
    {
        const int Port = 27125;
        async Task ToArena(string who, int mission)
        {

            for (int i = 0; i < 60 * 10 && Hub.InArena; i++) await ToSignal(GetTree(), SceneTree.SignalName.ProcessFrame);

    // WHAT A GUEST IS SHOWN. Both of these were host-only state read by drawing code, so on a
    // guest they were simply wrong -- and nothing said so, because the solo run is the host.
    async Task GuestSeesTheFight()
    {
        Boss boss = null;
        for (int i = 0; i < 60 * 2 && boss == null; i++)
        {   // InArena goes true before the boss node exists on this peer
            boss = H?.Boss;
            await ToSignal(GetTree(), SceneTree.SignalName.ProcessFrame);
        }
        Check(boss != null && !Net.Sim, "arena guest: the boss is here, and this peer is not the host");
        if (boss == null) return;

        // THE SUPER-MOVE BAR. _beam and _charge only tick under Net.Sim, so a guest's bar sat at
        // zero for the whole fight. Watched over time rather than sampled once: the failure was a
        // value that never moved, and a single reading cannot tell a stopped clock from a slow one.
        // Sampled only once a figure has actually ARRIVED. Before the first packet a guest's
        // countdown is -1, which reads as 0 s left, and a first sample of 0 makes "it went down"
        // impossible to tell from "it has not started".
        for (int i = 0; i < 60 * 3 && boss.NextSuperIn <= 0; i++) await ToSignal(GetTree(), SceneTree.SignalName.ProcessFrame);
        double firstFill = boss.SuperFill, firstLeft = boss.NextSuperIn;
        Check(firstLeft > 0, $"arena guest: the host's super-move countdown arrives at all ({firstLeft:0.0} s)");
        await Wait(1.5);
        Check(boss.NextSuperIn > 0 && boss.NextSuperIn < firstLeft - 0.5 && boss.SuperFill > firstFill,
              $"arena guest: the boss's super-move bar runs ({firstLeft:0.0} -> {boss.NextSuperIn:0.0} s, fill {firstFill:0.00} -> {boss.SuperFill:0.00})");
    }

    async Task Mp(string role)
    {

            H.Yard.Credits = 1000;                     // for the guest's purchase request
            Character.Exp = 0; Character.Level = 1; Character.BossCleared.Clear();
            Check(Net.I.Reachability == Net.Reach.Checking && Net.I.LanAddress.EndsWith(":27115"), "hosting starts at once; the router is asked in the background");
            await Wait(9.0);

            Check(Net.I.Reachability == Net.Reach.LanOnly && Net.I.LastStatus.Contains("forward UDP 27115") && Net.I.LastStatus.Contains("Tailscale"),
                  $"no UPnP router: LAN hosting stands, and the status explains the fix");

                  "carrier-grade and private addresses are recognised as unreachable from outside");
            // THE BUILD HANDSHAKE. We version save files; a SESSION is where two builds actually
            // get to disagree, and a guest on an older build would join and read every number
            // wrong. 1 as a literal, never Game.Version: a check that reads the number it is
            // testing cannot catch the number being wrong. The three ships below are the other
            // half of it -- the handshake has to admit real peers, not just refuse fake ones.
            Check(Net.Accepts(1) && Game.Version == 1 && !Net.Accepts(0) && !Net.Accepts(2),
                  $"a session admits this build and refuses any other (build {Game.Version})");
            var ships = Nodes<PlayerShip>().ToList();
            Check(ships.Count == 3, $"host sees 3 ships: itself and two guests ({ships.Count})");
            Check(ships.Any(s => s.Pilot == "Third"), "the second guest's identity reached the host");
            // hosting here (no router, no internet in the sandbox): both answers come back empty -> network only
            for (int i = 0; i < 60 * 9 && Net.I.Reachability == Net.Reach.Checking; i++) await ToSignal(GetTree(), SceneTree.SignalName.ProcessFrame);
            var smh = Nodes<SessionMenu>().First(); var rev = Nodes<Button>(smh).First(b => b.Name == "RevealAddress");
            Check(Net.I.Reachability == Net.Reach.LanOnly && !rev.Visible, $"no router or internet here: network only, nothing to reveal ({Net.I.Reachability})");
            Net.I.Reachable(Net.Reach.Manual, "203.0.113.7:27015", "Hosting (test).");
            await Wait(0.1);
            Check(rev.Visible && !rev.Text.Contains("203.0.113.7") && !Net.I.LastStatus.Contains("203.0.113.7"), "the address friends need is hidden until clicked");
            rev.EmitSignal(BaseButton.SignalName.Pressed); await Wait(0.05);
            Check(rev.Text.Contains("203.0.113.7:27015"), "click: it shows the public address and port");
            rev.EmitSignal(BaseButton.SignalName.Pressed); await Wait(0.05);
            Check(!rev.Text.Contains("203.0.113.7"), "click again: hidden");
            Character.BossCleared.Clear();
            H.AnnounceBossKill(1);                // every pilot: its own EXP by its own level
            H.SpawnRaider(new Vector2(-3500, 3500));   // far off: the guests must see it
            H.SpawnRaider(new Vector2(-3700, 3700), RaiderKind.Heavy);
            Check(!H.AllReady, "the host cannot warp until every pilot is ready");
            H.SetMyReady(true); Me.AutopilotTo = null;
            for (int i = 0; i < 600 && !H.AllReady; i++) await ToSignal(GetTree(), SceneTree.SignalName.ProcessFrame);
            Check(H.AllReady && H.PartyIds.Count() == 3, "both guests' READY reached the host: the party of three is ready");
            Check(Multiplayer.GetPeers().All(id => Hub.PeerSector(id) == Hub.SectorKind.Home),
                  "every guest has told the host it is home (the yard talks only to guests at home)");
            H.StartMission();
            Check(Character.Exp == 550, $"the host gets its own EXP for the kill (level 1: 550; exp {Character.Exp})");
            Check(H.Yard.Level("salvage_rate") == 1 && H.Yard.Credits == 900, "a guest's BUY request was honoured by the host");
            var g = ships.FirstOrDefault(s => s.Pilot == "Guesty");
            Check(g != null && g.Name == PlayerShip.NodeName(g.OwnerId), "guest ship named by its id");
            Check(g != null && g.Pilot == "Guesty" && g.Class == ShipClass.Carrier && g.Main == new Color(1, 0, 0), $"guest identity reached host ({g?.Pilot}/{g?.Class}/{g?.Main})");
            var dummy = H.GetNode<TargetDummy>("TargetDummy1");
            Check(g != null && g.WingTarget == dummy, "guest's Space order reached the host");
            Check(dummy.Total > 5, $"guest's wing hit the dummy on the host ({dummy.Total:0.0})");
            Check(g != null && g.PdActive, "guest's PD activation reached the host");
            Check(Nodes<Torpedo>().Any(x => !x.Cosmetic) || dummy.Total > 12, "guest's bomber strike launched real torpedoes on the host");
            // stay up while the guest watches its bombers come home and dock (it checks ~22 s in)
            await Wait(15.0);
            Net.I.GoOffline();   // drop the guest
            await Wait(2.0);
            for (int i = 0; i < 60 * 4 && !Net.I.NetworkIdle; i++) await ToSignal(GetTree(), SceneTree.SignalName.ProcessFrame);
            Check(Net.I.NetworkIdle, "back offline: no socket, no lookup, no router job left running");
        }
        else if (role == "guest2")
        {   // a third player: joins, sees everyone, leaves with the session
            await Wait(1.6);
            Net.I.Join("127.0.0.1:27115");
            await Wait(1.2);
            Character.Name = "Third"; Character.Class = ShipClass.Battleship;
            Priv(H, "ApplyLocalIdentity"); Priv(H, "SendIdentity", 0);
            await Wait(6.0);
            H.SetMyReady(true);                           // READY for the bounty
            Nodes<PlayerShip>().First(x => x.Mine).AutopilotTo = null;   // stay put: the scene must not change mid-test
            var all = Nodes<PlayerShip>().ToList();
            Check(all.Count == 3 && all.Any(s => s.Pilot == "HostPilot") && all.Any(s => s.Pilot == "Guesty"),
                  $"third player sees the host and the other guest ({string.Join(", ", all.Select(s => s.Pilot))})");
            Check(all.Count(s => s.Mine) == 1 && all.First(s => s.Mine).OwnerId == Net.LocalId, "third player flies its own ship");
            await Wait(22.0);
            Check(!Net.IsOnline, "third player dropped to offline when the host left");
        }
        else
        {
            H.Yard.Ore = 5000;       // "own" hub total, should come back after leaving
            // a NON-default loadout (two chips off): a host that ignored the guest's gear would fall
            // back to the default (five chips) and replicate 250 hull instead of 230
            var gl = Character.LoadoutFor(Character.Class); gl[8] = ""; gl[9] = "";
            await Wait(0.8);
            Net.I.Join("127.0.0.1:27115");
            await Wait(1.2);
            Character.Name = "Guesty"; Character.Main = new Color(1, 0, 0); Character.Class = ShipClass.Carrier;
            Priv(H, "ApplyLocalIdentity"); Priv(H, "SendIdentity", 0);
            var mine = Nodes<PlayerShip>().FirstOrDefault(s => s.Mine);
            Check(mine != null && mine.OwnerId == Net.LocalId && Net.LocalId != 1, $"guest owns its own ship (id {Net.LocalId})");
            Check(Nodes<PlayerShip>().Count() >= 2, $"guest sees the host ({Nodes<PlayerShip>().Count()} ships)");
            H.Yard.BuyUpgrade("salvage_rate");            // a request: the host decides
            H.SetMyReady(true);                           // READY for the bounty
            Nodes<PlayerShip>().First(x => x.Mine).AutopilotTo = null;   // stay put: the scene must not change mid-test
            var dummy = H.GetNode<TargetDummy>("TargetDummy1");
            if (mine != null) { mine.Position = dummy.Position + new Vector2(-300, 300); mine.Velocity = Vector2.Zero; }
            await Wait(1.5);
            Key(global::Godot.Key.Tab); await Wait(0.05);
            Check(H.Selected == dummy, "guest: Tab selects the nearest dummy");
            Key(global::Godot.Key.Space);          // fighters attack
            Key(global::Godot.Key.F);              // bomber strike
            await Wait(1.0);
            Key(global::Godot.Key.Q);              // point defence
            // torpedoes live a couple of seconds; watch for one across the window
            bool sawTorp = false;
            for (int i = 0; i < 240; i++) { if (Nodes<Torpedo>().Any(x => x.Cosmetic)) sawTorp = true; await ToSignal(GetTree(), SceneTree.SignalName.ProcessFrame); }
            Check(mine != null && mine.PdActive, "guest sees its PD window running (replicated)");
            Check(sawTorp, "guest sees cosmetic copies of its torpedoes");
            var host = Nodes<PlayerShip>().FirstOrDefault(s => s.OwnerId == 1);
            Check(Nodes<PlayerShip>().Count() == 3 && Nodes<PlayerShip>().Any(s => s.Pilot == "Third"), "guest sees the other guest too (3 ships)");
            Check(H.Yard.Level("salvage_rate") == 1, "the upgrade the host granted reached the guest");
            for (int i = 0; i < 480 && Character.Exp != 550; i++) await ToSignal(GetTree(), SceneTree.SignalName.ProcessFrame);
            Check(Character.Exp == 550, $"the host's kill reached this guest, which took its own 550 (exp {Character.Exp})");
            for (int i = 0; i < 60 * 6 && H.Raiders.Count == 0; i++) await ToSignal(GetTree(), SceneTree.SignalName.ProcessFrame);
            for (int i = 0; i < 60 * 3 && !H.Raiders.Any(r => r.Heavy); i++) await ToSignal(GetTree(), SceneTree.SignalName.ProcessFrame);
            Check(H.Raiders.Count >= 2 && Combat.Hostiles.OfType<Raider>().Any() && H.Raiders.Any(r => r.Heavy),
                  "the host's raiders reached this guest, a heavy as a heavy (hostiles it can target)");
            var gm = H.Yard.Gatherers.First(x => x.Kind == GatherKind.Miner);
            Check(H.Yard.Gatherers.Any(x => x.Position.DistanceTo(Hub.BasePos) > 400) && H.Yard.Hauler.Position.Y == Hub.LaneY,
                  "guest sees the host's miner out working and the hauler on its lane");
            Check(host != null && host.Pilot == "HostPilot", $"host identity reached guest ({host?.Pilot})");
            Check(H.Yard.Ore < 1000, $"guest shows host's totals ({H.Yard.Ore:0})");
            Check(dummy.Total > 0 && dummy.LastSecond > 0, $"dummy readout reached the guest ({dummy.LastSecond:0.00} / {dummy.Total:0.0})");
            int near = Nodes<Wing>().Count(w => w.Carrier == mine && w.Kind == WingKind.Fighter && w.Position.DistanceTo(dummy.Position) < 400);
            Check(near >= 3, $"guest sees its fighters at the target, as the host flies them ({near})");
            // a flash lives 0.1 s, so watch for one over two seconds rather than sampling once
            var flashes = (System.Collections.IList)H.GetType().GetField("_flashes", BindingFlags.NonPublic | BindingFlags.Instance).GetValue(H);
            int seen = 0;
            for (int i = 0; i < 60; i++) { if (flashes.Count > 0) seen++; await ToSignal(GetTree(), SceneTree.SignalName.ProcessFrame); }
            Check(seen > 0, $"guest sees the host's hit flashes ({seen} of 60 frames)");
            Check(mine != null && Math.Abs(mine.MaxHp - 230) < 1e-6 && Math.Abs(mine.Hp - 230) < 1e-6,
                  $"hull replicated from the host, which applied this pilot's own gear (three chips: 200 x 1.15 = 230; {mine?.Hp:0})");
            await Wait(12.0);                      // strike done, bombers home and docked
            var gb = Nodes<Wing>().Where(w => w.Carrier == mine && w.IsBomber).ToArray();
            Check(gb.Length == 2 && gb.All(w => w.Docked && w.GlobalPosition.DistanceTo(mine.DockSlot(w).pos) < 8f),
                  "guest sees its bombers docked on the carrier (state and position from the host)");
            Check(H.GetNodeOrNull("MissionPortal") != null && H.Mission == Hub.MissionState.PortalOpen,
                  "the host's mission portal opened here too (the party was ready)");
            await Wait(6.0);                       // the host closes the session ~24.5 s in
            Check(!Net.IsOnline && Net.LocalId == 1, "guest fell back offline when host left");
            // Back to its OWN world, with nothing lost: everything held -- stock, the
            // ships' cargo, and credits from a sale -- is at least the 5000 it had, and
            // nowhere near the host's few hundred.
            double own = H.Yard.Ore + H.Yard.Salvage + H.Yard.Hauler.Cargo + H.Yard.Gatherers.Sum(x => x.Cargo) + H.Yard.Credits;
            double share = Missions.BountyEach(1, 3);                // the host's announced kill, a party of three
            Check(own - share >= 5000 - 1e-6 && own - share < 5400, $"guest's own totals restored, nothing lost, plus its bounty share ({own:0.00} held, {share:0.00} of it the share)");
            Check(H.Yard.Level("salvage_rate") == 0, "guest's own upgrade levels restored (the host's level 1 stays in the host's world)");
            Check(H.Yard.Hauler.Cargo <= H.Yard.Capacity && H.Yard.Gatherers.All(x => x.Cargo == 0 || x.State == Gatherer.St.Working),
                  "economy ships restart docked and empty in your own world");
            var ships = Nodes<PlayerShip>().ToList();
            Check(ships.Count == 1 && ships[0].Mine && ships[0].Name == "Ship_1", "guest rebuilt to its own single ship");
        }
    }
